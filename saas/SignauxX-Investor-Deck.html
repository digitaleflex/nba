Schema Prisma — 1 model + 22 FK

model Organization {
  id String @id @default(cuid())
  slug String @unique // coach1
  name String // "Coach Karim"
  ownerId String
  plan Tier // STARTER/PRO/SCALE
  stripeCustomerId String?
  signals Signal[]
  members User[]
}

// Sur chaque table:
model Signal {
  orgId String // + index
  @@index([orgId, createdAt])
}
